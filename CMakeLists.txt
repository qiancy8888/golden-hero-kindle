cmake_minimum_required(VERSION 3.12)
project(kindle_golden_hero LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
option(GOLDEN_PLATINUM "Build the GBC-based Platinum Hero port" OFF)
if(GOLDEN_PLATINUM)
  enable_language(C)
  include(cmake/Platinum.cmake)
  return()
endif()
option(GOLDEN_STANDALONE "Build the ROM-independent game" OFF)
if(GOLDEN_STANDALONE)
  add_library(golden_opening game/town.cpp game/view.cpp)
  target_include_directories(golden_opening PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/game)
  target_compile_options(golden_opening PRIVATE -Wall -Wextra)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(golden_hero_standalone kindle/standalone.cpp kindle/fb.cpp kindle/touch.cpp)
    target_compile_options(golden_hero_standalone PRIVATE -Wall -Wextra)
    target_link_libraries(golden_hero_standalone PRIVATE golden_opening rt)
  endif()
  include(CTest)
  if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_executable(test_opening tests/test_opening.cpp)
    target_link_libraries(test_opening PRIVATE golden_opening)
    add_test(NAME standalone_opening COMMAND test_opening)
  endif()
  return()
endif()
option(GOLDEN_MOCK_CORE "Compile with mock core for host adapter tests" OFF)
if(GOLDEN_MOCK_CORE)
  set(CORE tests/mock_nc1020.cpp)
else()
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/upstream/nc1020.cpp")
    message(FATAL_ERROR "Missing upstream NC1020 source. Run python3 scripts/fetch_upstream.py first.")
  endif()
  set(CORE upstream/nc1020.cpp)
endif()
add_executable(golden_hero kindle/main.cpp kindle/display_test.cpp kindle/fb.cpp kindle/touch.cpp kindle/demo.cpp ${CORE})
target_include_directories(golden_hero PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}/kindle ${CMAKE_CURRENT_SOURCE_DIR}/upstream
  ${CMAKE_CURRENT_SOURCE_DIR}/tests)
target_compile_options(golden_hero PRIVATE -O2 -Wall -Wextra -Wno-register)
# Keep librt for compatibility with older Kindle glibc toolchains.
target_link_libraries(golden_hero PRIVATE rt)
